从任何一个 Gateway 接入。

Client 创建任务，Gateway 将请求路由到目标分区，让接入与任务处理各司其职。

每个分区独立选主，多副本共同保护任务状态。

GO · 已配置并连接 client
// 先创建订阅，等待分发至 Broker 分区
_, err := client.NewCreateJobSubscriptionCommand().
    JobType("payment").Send(ctx)
if err != nil { log.Fatal(err) }

// 确认分发完成后，创建普通任务
job, err := client.NewCreateJobCommand().
    JobType("payment").
    Variables(map[string]any{"orderId": 42}).
    Send(ctx)
if err != nil { log.Fatal(err) }

TERMINAL · 本地集群已运行
cd clients/go
go build -o bin/tkctl ./cmd/tkctl
export TASKITE_CLUSTER_ID=demo

bin/tkctl create subscription payment
# 确认订阅已分发后创建任务
bin/tkctl create job payment --variables '{"orderId":42}'
bin/tkctl create timer payment --type duration --expression PT30S
bin/tkctl activate jobs payment --handler cat
